# Build and run the C and C++ examples against the wickra-screener C ABI.
#
#   cargo build --release -p wickra-screener-c
#   cmake -S examples/c -B examples/c/build
#   cmake --build examples/c/build --config Release
#   ctest --test-dir examples/c/build -C Release --output-on-failure
#
# WICKRA_SCREENER_LIB_DIR defaults to the workspace target/release directory and
# can be overridden for out-of-tree builds.

cmake_minimum_required(VERSION 3.15)
project(wickra_screener_examples C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# The C++ wrapper (bindings/c/include/wickra_screener.hpp) is C++17.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(WICKRA_SCREENER_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release"
    CACHE PATH "Directory containing the built wickra-screener C ABI library")

set(WICKRA_SCREENER_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../bindings/c/include")

find_library(WICKRA_SCREENER_LIB
    NAMES wickra_screener libwickra_screener
    HINTS "${WICKRA_SCREENER_LIB_DIR}"
    REQUIRED)

add_executable(scan scan.c)
target_include_directories(scan PRIVATE "${WICKRA_SCREENER_INCLUDE}")
target_link_libraries(scan PRIVATE "${WICKRA_SCREENER_LIB}")

add_executable(scanpp scan.cpp)
target_include_directories(scanpp PRIVATE "${WICKRA_SCREENER_INCLUDE}")
target_link_libraries(scanpp PRIVATE "${WICKRA_SCREENER_LIB}")

# Streaming equals batch, once through the raw ABI and once through the C++
# hull. Both drive the two-call length protocol, which is where a mutating
# command used to run twice.
add_executable(streaming_test streaming_test.c)
target_include_directories(streaming_test PRIVATE "${WICKRA_SCREENER_INCLUDE}")
target_link_libraries(streaming_test PRIVATE "${WICKRA_SCREENER_LIB}")

add_executable(streaming_test_pp streaming_test.cpp)
target_include_directories(streaming_test_pp PRIVATE "${WICKRA_SCREENER_INCLUDE}")
target_link_libraries(streaming_test_pp PRIVATE "${WICKRA_SCREENER_LIB}")

# Golden parity from C. There is no portable directory API in C, so the spec
# list is globbed here and written into a generated header -- which keeps the
# property the other bindings get from a runtime glob: a spec added to the
# corpus is covered without editing the test. CONFIGURE_DEPENDS makes the build
# system re-glob when the directory changes, so a new spec does not need a
# manual re-configure to be noticed.
set(GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../golden")
file(GLOB GOLDEN_SPEC_FILES CONFIGURE_DEPENDS "${GOLDEN_DIR}/specs/*.json")
set(GOLDEN_SPEC_ENTRIES "")
foreach(spec_file ${GOLDEN_SPEC_FILES})
    get_filename_component(spec_name "${spec_file}" NAME)
    string(APPEND GOLDEN_SPEC_ENTRIES "    \"${spec_name}\",
")
endforeach()
list(LENGTH GOLDEN_SPEC_FILES GOLDEN_SPEC_COUNT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/golden_specs.h"
"/* Generated by examples/c/CMakeLists.txt -- do not edit. */
"
"#ifndef WICKRA_SCREENER_GOLDEN_SPECS_H
"
"#define WICKRA_SCREENER_GOLDEN_SPECS_H
"
"#include <stddef.h>
"
"#define GOLDEN_DIR \"${GOLDEN_DIR}\"
"
"static const char *GOLDEN_SPECS[] = {
${GOLDEN_SPEC_ENTRIES}};
"
"static const size_t GOLDEN_SPEC_COUNT = ${GOLDEN_SPEC_COUNT};
"
"#endif
")

add_executable(golden_test golden_test.c)
target_include_directories(golden_test PRIVATE
    "${WICKRA_SCREENER_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(golden_test PRIVATE "${WICKRA_SCREENER_LIB}")

# On Windows the runtime DLL must sit next to the executable (there is no rpath);
# without it the loader pops a modal "DLL not found" dialog that hangs ctest. Copy
# it next to each example after build. (Unix resolves the .so/.dylib via the
# linker's embedded path, so this is Windows-only.)
if(WIN32)
    foreach(example scan scanpp streaming_test streaming_test_pp golden_test)
        add_custom_command(TARGET ${example} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${WICKRA_SCREENER_LIB_DIR}/wickra_screener.dll"
                "$<TARGET_FILE_DIR:${example}>"
            VERBATIM)
    endforeach()
endif()

enable_testing()
add_test(NAME scan COMMAND scan)
add_test(NAME scanpp COMMAND scanpp)
add_test(NAME streaming_test COMMAND streaming_test)
add_test(NAME streaming_test_pp COMMAND streaming_test_pp)
add_test(NAME golden_test COMMAND golden_test)
# A missing runtime dependency must fail fast, never block the runner for hours.
set_tests_properties(scan scanpp streaming_test streaming_test_pp golden_test
    PROPERTIES TIMEOUT 60)
